docker run --rm -v "$PWD/php8.4-debs:/out" ubuntu:24.04 bash -c '
set -e

rm -f /etc/apt/apt.conf.d/docker-clean

apt-get update

apt-get install -y --no-install-recommends \
  software-properties-common \
  ca-certificates \
  gnupg \
  curl

add-apt-repository ppa:ondrej/php -y

apt-get update

apt-get install -y --download-only --no-install-recommends \
  php8.4 \
  php8.4-common \
  php8.4-cli \
  php8.4-fpm \
  php8.4-opcache \
  php8.4-readline \
  php8.4-xml \
  php8.4-sqlite3 \
  php8.4-redis \
  php8.4-zip \
  php8.4-curl \
  php8.4-mbstring \
  php8.4-mysql

cp /var/cache/apt/archives/*.deb /out/
'

mkdir -p php8.3-debs

docker run --rm -v "$PWD/php8.3-debs:/out" ubuntu:24.04 bash -c '
set -e

rm -f /etc/apt/apt.conf.d/docker-clean

apt-get update

apt-get install -y --download-only --no-install-recommends \
  php8.3 \
  php8.3-common \
  php8.3-cli \
  php8.3-fpm \
  php8.3-opcache \
  php8.3-readline \
  php8.3-xml \
  php8.3-sqlite3 \
  php8.3-zip \
  php8.3-curl \
  php8.3-mbstring \
  php8.3-mysql

cp /var/cache/apt/archives/*.deb /out/
'



FROM ubuntu:24.04

ENV DEBIAN_FRONTEND=noninteractive
ENV PHP_VERSION=8.4

COPY php8.4-debs/*.deb /tmp/php8.4/

RUN dpkg -i /tmp/php8.4/*.deb || true && \
    apt-get install -f -y && \
    ln -sf /etc/php/${PHP_VERSION}/fpm /etc/php-fpm && \
    ln -sf /usr/sbin/php-fpm${PHP_VERSION} /usr/sbin/php-fpm && \
    rm -rf /tmp/php8.4

CMD ["bash"]
